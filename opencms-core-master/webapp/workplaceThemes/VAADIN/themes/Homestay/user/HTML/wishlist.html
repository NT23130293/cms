// Mã mẫu xử lý Toggle / Xóa item khỏi Wishlist
function toggleWishlistItem(homestayId) {
  const list = JSON.parse(localStorage.getItem('tourist_wishlist') || '[]');
  const nextList = list.includes(homestayId) ? list.filter(id => id !== homestayId) : [...list, homestayId];
  localStorage.setItem('tourist_wishlist', JSON.stringify(nextList));
  renderWishlistUI(nextList);
}
